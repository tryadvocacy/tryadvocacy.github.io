<script>
document.addEventListener('DOMContentLoaded', () => {

    const apiKey = "";
    const apiUrl = "https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-05-20:generateContent?key=" + apiKey;

    const explanationModal = document.getElementById('explanation-modal');
    const closeButton = document.getElementById('close-modal');
    const explanationTitle = document.getElementById('explanation-title');
    const explanationText = document.getElementById('explanation-text');
    const explanationLoader = document.getElementById('explanation-loader');
    const copyExplanationBtn = document.getElementById('copy-explanation');
    
    closeButton.addEventListener('click', () => {
        explanationModal.classList.add('hidden');
    });

    copyExplanationBtn.addEventListener('click', () => {
        const textToCopy = explanationText.textContent;
        navigator.clipboard.writeText(textToCopy).then(() => {
            alert('Explanation copied to clipboard!');
        }).catch(err => {
            console.error('Failed to copy text: ', err);
            alert('Failed to copy text.');
        });
    });

    async function generateSimpleExplanation(name, description) {
        explanationModal.classList.remove('hidden');
        explanationLoader.classList.remove('hidden');
        explanationTitle.textContent = "Simplifying the " + name + " Function";
        explanationText.textContent = "";

        const systemPrompt = "You are an expert cybersecurity tutor. Explain the following NIST CSF 2.0 function in simple, easy-to-understand terms for a non-technical audience. Use analogies and avoid jargon. Be concise and encouraging.";
        const userQuery = `Simplify the explanation for the "${name}" function. Its core description is: "${description}"`;

        const payload = {
            contents: [{ parts: [{ text: userQuery }] }],
            systemInstruction: { parts: [{ text: systemPrompt }] },
        };

        try {
            const response = await fetch(apiUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });
            const result = await response.json();
            const text = result.candidates[0].content.parts[0].text;
            explanationText.textContent = text;
        } catch (error) {
            explanationText.textContent = "Failed to generate explanation. Please try again.";
            console.error(error);
        } finally {
            explanationLoader.classList.add('hidden');
        }
    }

    const functionsData = [
        { name: 'Govern (GV)', icon: '🏛️', description: "The newest function, it establishes the foundation for cybersecurity risk management. It's about how an organization's cybersecurity strategy, expectations, and policies are established, communicated, and monitored. It ensures cybersecurity is a strategic business decision." },
        { name: 'Identify (ID)', icon: '🔍', description: "This function focuses on understanding your current cybersecurity risks. It involves identifying your assets, systems, data, and capabilities to understand what you need to protect." },
        { name: 'Protect (PR)', icon: '🛡️', description: "This function details the safeguards you use to manage cybersecurity risks. It includes implementing controls to prevent or reduce the likelihood and impact of a cybersecurity incident." },
        { name: 'Detect (DE)', icon: '📡', description: "This function is about discovering and analyzing possible cybersecurity attacks and compromises. It includes continuously monitoring for anomalies and suspicious events." },
        { name: 'Respond (RS)', icon: '🚒', description: "This function covers the actions you take after a cybersecurity incident is detected. It involves containing the event, analyzing its cause, and communicating with stakeholders." },
        { name: 'Recover (RC)', icon: '🔄', description: "This function focuses on restoring assets and operations that were affected by a cybersecurity incident. It ensures a quick return to normal business operations." }
    ];

    const functionsContainer = document.getElementById('functions-container');
    functionsData.forEach(func => {
        const card = document.createElement('div');
        card.className = 'bg-white p-6 rounded-lg shadow-sm cursor-pointer hover:shadow-lg transition-shadow';
        card.innerHTML = `
            <div class="flex items-center mb-3">
                <span class="text-3xl mr-4">${func.icon}</span>
                <h3 class="text-lg font-semibold text-gray-900">${func.name}</h3>
            </div>
            <p class="description text-gray-600 hidden">${func.description}</p>
            <button class="explain-btn mt-4 inline-flex items-center px-3 py-1.5 border border-transparent text-xs font-medium rounded-full shadow-sm text-white bg-blue-600 hover:bg-blue-700 focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-blue-500 transition-colors">
                <span class="mr-1">✨</span> Explain simply
            </button>
        `;
        functionsContainer.appendChild(card);
        card.querySelector('.explain-btn').addEventListener('click', (e) => {
            e.stopPropagation();
            generateSimpleExplanation(func.name, func.description);
        });
        card.addEventListener('click', () => {
            card.querySelector('.description').classList.toggle('hidden');
            card.classList.toggle('bg-blue-50');
        });
    });

    const ctx = document.getElementById('functionsChart').getContext('2d');
    new Chart(ctx, {
        type: 'radar',
        data: {
            labels: functionsData.map(f => f.name.split(' ')[0]),
            datasets: [{
                label: 'Core Functions',
                data: [1, 1, 1, 1, 1, 1],
                fill: true,
                backgroundColor: 'rgba(59, 130, 246, 0.2)',
                borderColor: 'rgb(59, 130, 246)',
                pointBackgroundColor: 'rgb(59, 130, 246)',
                pointBorderColor: '#fff',
                pointHoverBackgroundColor: '#fff',
                pointHoverBorderColor: 'rgb(59, 130, 246)'
            }]
        },
        options: {
            maintainAspectRatio: false,
            scales: {
                r: {
                    angleLines: { display: true },
                    suggestedMin: 0,
                    suggestedMax: 1.2,
                    pointLabels: { font: { size: 14 } },
                    ticks: { display: false }
                }
            },
            plugins: {
                legend: { display: false }
            }
        }
    });

    const tiersData = [
        { name: 'Tier 1: Partial', description: "Risk management practices are informal, and cybersecurity activities may be reactive." },
        { name: 'Tier 2: Risk-Informed', description: "Risk management practices are approved by management, and activities are based on an organization's risk-informed approach." },
        { name: 'Tier 3: Repeatable', description: "The organization has repeatable and formal processes for managing cybersecurity risk." },
        { name: 'Tier 4: Adaptive', description: "The organization has a highly mature, proactive, and adaptive approach to cybersecurity that evolves with the threat landscape." }
    ];
    let selectedTierName = null;

    const tierStepper = document.getElementById('tier-stepper');
    const tierDescription = document.getElementById('tier-description');
    const adviceContainer = document.getElementById('advice-container');
    const adviceText = document.getElementById('advice-text');
    const adviceLoader = document.getElementById('advice-loader');
    const copyAdviceBtn = document.getElementById('copy-advice');
    
    copyAdviceBtn.addEventListener('click', () => {
        const textToCopy = adviceText.textContent;
        navigator.clipboard.writeText(textToCopy).then(() => {
            alert('Advice copied to clipboard!');
        }).catch(err => {
            console.error('Failed to copy text: ', err);
            alert('Failed to copy text.');
        });
    });

    async function generateAdvice(tierName) {
        adviceContainer.classList.remove('hidden');
        adviceLoader.classList.remove('hidden');
        adviceText.textContent = "";
        copyAdviceBtn.classList.add('hidden');

        const systemPrompt = "You are a cybersecurity consultant with expertise in the NIST CSF. Your task is to provide three short, actionable pieces of advice for an organization to improve its cybersecurity posture, specifically targeting the next maturity tier. The advice should be practical and easy to understand.";
        const userQuery = `The organization is currently at the maturity level "${tierName}". What three actions should it take to improve?`;

        const payload = {
            contents: [{ parts: [{ text: userQuery }] }],
            systemInstruction: { parts: [{ text: systemPrompt }] },
        };

        try {
            const response = await fetch(apiUrl, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(payload)
            });
            const result = await response.json();
            const text = result.candidates[0].content.parts[0].text;
            adviceText.textContent = text;
            copyAdviceBtn.classList.remove('hidden');
        } catch (error) {
            adviceText.textContent = "Failed to generate advice. Please try again.";
            console.error(error);
        } finally {
            adviceLoader.classList.add('hidden');
        }
    }

    tiersData.forEach((tier, index) => {
        const step = document.createElement('button');
        step.className = 'tier-step text-left w-full p-3 rounded-md hover:bg-gray-100 transition-colors focus:outline-none focus:ring-2 focus:ring-blue-500';
        step.innerHTML = `<span class="font-semibold text-gray-700">${tier.name}</span>`;
        step.dataset.index = index;
        tierStepper.appendChild(step);
    });

    const getAdviceButton = document.createElement('button');
    getAdviceButton.className = 'mt-4 inline-flex items-center px-4 py-2 border border-transparent text-sm font-medium rounded-full shadow-sm text-white bg-blue-600 hover:bg-blue-700 focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-blue-500 transition-colors';
    getAdviceButton.innerHTML = `<span class="mr-1">✨</span> Get implementation advice`;
    getAdviceButton.style.display = 'none';

    tierStepper.appendChild(getAdviceButton);

    tierStepper.addEventListener('click', (e) => {
        const step = e.target.closest('.tier-step');
        if (step) {
            const index = step.dataset.index;
            selectedTierName = tiersData[index].name;
            tierDescription.innerHTML = `<h4 class="font-bold text-blue-800 mb-2">${tiersData[index].name}</h4><p class="text-blue-900">${tiersData[index].description}</p>`;
            document.querySelectorAll('.tier-step').forEach(s => s.classList.remove('bg-blue-100', 'text-blue-800'));
            step.classList.add('bg-blue-100', 'text-blue-800');
            getAdviceButton.style.display = 'inline-flex';
        }
    });
    
    getAdviceButton.addEventListener('click', () => {
        if (selectedTierName) {
            generateAdvice(selectedTierName);
        }
    });

    const implementationData = [
        { name: '1. Prioritize & Scope', details: "Define the scope of your cybersecurity program and identify your mission-critical objectives." },
        { name: '2. Orient', details: "Understand your current legal, regulatory, and contractual requirements." },
        { name: '3. Create Current Profile', details: "Assess your current cybersecurity practices and map them to the CSF 2.0's subcategories." },
        { name: '4. Conduct Risk Assessment', details: "Analyze the threats and vulnerabilities to your organization." },
        { name: '5. Create Target Profile', details: "Define your desired cybersecurity posture based on your risk assessment and business needs." },
        { name: '6. Analyze Gaps', details: "Compare your Current and Target Profiles to find gaps. Prioritize them based on risk." },
        { name: '7. Implement Action Plan', details: "Use CSF 2.0 resources to create and execute a plan to close the gaps." },
        { name: '8. Monitor Progress', details: "Continuously review and update your Current Profile as you improve your cybersecurity posture." }
    ];

    const implStepsContainer = document.getElementById('implementation-steps');
    const implDetailsContainer = document.getElementById('implementation-details');

    implementationData.forEach((step, index) => {
        const button = document.createElement('button');
        button.className = 'impl-step bg-white border border-gray-300 text-gray-700 font-medium py-2 px-4 rounded-lg hover:bg-gray-50 transition-colors focus:outline-none focus:ring-2 focus:ring-blue-500';
        button.textContent = step.name;
        button.dataset.index = index;
        implStepsContainer.appendChild(button);
    });

    implStepsContainer.addEventListener('click', (e) => {
        const step = e.target.closest('.impl-step');
        if (step) {
            const index = step.dataset.index;
            implDetailsContainer.innerHTML = `<h4 class="font-bold text-gray-800 mb-2">${implementationData[index].name}</h4><p class="text-gray-700">${implementationData[index].details}</p>`;
            document.querySelectorAll('.impl-step').forEach(s => s.classList.remove('bg-blue-600', 'text-white', 'border-blue-600'));
            step.classList.add('bg-blue-600', 'text-white', 'border-blue-600');
        }
    });

    const resourcesData = [
        { name: 'NIST CSF 2.0 Official Publication', description: 'The definitive guide to the framework. Find it on the NIST website.' },
        { name: 'CSF 2.0 Online Reference Tool', description: 'A searchable, online version of the Core for easy navigation.' },
        { name: 'Implementation Examples', description: 'NIST provides examples of how to achieve specific subcategories.' },
        { name: 'Quick Start Guides', description: 'Tailored guides for specific audiences, like small businesses.' },
        { name: 'Informative References', description: 'A catalog that maps CSF outcomes to over 50 other cybersecurity standards.' },
        { name: 'Training and Certifications', description: 'Courses from organizations like ISC2 to demonstrate expertise.' }
    ];
    
    const resourcesList = document.getElementById('resources-list');
    resourcesData.forEach(resource => {
        const item = document.createElement('li');
        item.className = 'border-l-4 border-teal-500 pl-4 py-2';
        item.innerHTML = `
            <h4 class="font-semibold text-gray-800">${resource.name}</h4>
            <p class="text-gray-600">${resource.description}</p>
        `;
        resourcesList.appendChild(item);
    });
    
    document.querySelectorAll('.nav-link').forEach(anchor => {
        anchor.addEventListener('click', function (e) {
            e.preventDefault();
            document.querySelector(this.getAttribute('href')).scrollIntoView({
                behavior: 'smooth'
            });
        });
    });
});
